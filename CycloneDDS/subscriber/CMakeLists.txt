cmake_minimum_required(VERSION 3.16)
project(xr_subscriber_native LANGUAGES C)

find_package(CycloneDDS CONFIG REQUIRED)
set(PLUGIN "${CMAKE_CURRENT_SOURCE_DIR}/../cdds_plugin")
file(GLOB_RECURSE DDS_TYPES CONFIGURE_DEPENDS "${PLUGIN}/generated/*.c")
add_library(cdds_plugin SHARED "${PLUGIN}/src/cdds_plugin.c" ${DDS_TYPES})
target_include_directories(cdds_plugin PRIVATE "${PLUGIN}/include" "${PLUGIN}/generated")
target_compile_features(cdds_plugin PRIVATE c_std_99)
target_link_libraries(cdds_plugin PRIVATE CycloneDDS::ddsc)
if(WIN32)
    target_link_libraries(cdds_plugin PRIVATE ws2_32 iphlpapi bcrypt)
    install(FILES "$<TARGET_FILE:CycloneDDS::ddsc>" DESTINATION .)
endif()
set_target_properties(cdds_plugin PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS cdds_plugin LIBRARY DESTINATION . RUNTIME DESTINATION .)
